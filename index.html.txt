    <script>
        // Tab Navigation
        function switchTab(tab) {
            ['calculators', 'properties', 'tracker'].forEach(t => {
                document.getElementById(`content-${t}`).classList.add('hidden');
                document.getElementById(`tab-${t}`).className = "py-2 rounded-lg text-slate-600 transition";
            });
            document.getElementById(`content-${tab}`).classList.remove('hidden');
            document.getElementById(`tab-${tab}`).className = "py-2 rounded-lg bg-white shadow-sm text-slate-900 transition font-semibold";
            if(tab === 'tracker') { setTimeout(() => { map.invalidateSize(); }, 200); }
        }

        function switchCalcSub(sub) {
            ['tdsr', 'proceeds', 'purchase'].forEach(s => {
                document.getElementById(`calc-${s}`).classList.add('hidden');
                document.getElementById(`subtab-${s}`).className = "text-slate-500 pb-1";
            });
            document.getElementById(`calc-${sub}`).classList.remove('hidden');
            document.getElementById(`subtab-${sub}`).className = "text-blue-600 border-b-2 border-blue-600 pb-1 font-semibold";
        }

        // Calculations
        function calculateTDSR() {
            const income = parseFloat(document.getElementById('tdsr-income').value);
            const debt = parseFloat(document.getElementById('tdsr-debt').value);
            const tenure = parseInt(document.getElementById('tdsr-tenure').value);
            const rate = parseFloat(document.getElementById('tdsr-rate').value) / 100;

            const maxAllowedDebtService = income * 0.55;
            const availableForMortgageTDSR = maxAllowedDebtService - debt;
            const availableForMortgageMSR = income * 0.30;

            // Monthly mortgage formula: M = P [ r(1+r)^n ] / [ (1+r)^n – 1]
            const monthlyRate = rate / 12;
            const months = tenure * 12;
            const calculateLoan = (pmt) => pmt * (Math.pow(1 + monthlyRate, months) - 1) / (monthlyRate * Math.pow(1 + monthlyRate, months));

            const maxLoanTDSR = calculateLoan(Math.max(0, availableForMortgageTDSR));
            const maxLoanMSR = calculateLoan(Math.max(0, availableForMortgageMSR));

            document.getElementById('res-tdsr-limit').innerText = `$${maxAllowedDebtService.toLocaleString('en-US', {maximumFractionDigits:0})}`;
            document.getElementById('res-msr-limit').innerText = `$${availableForMortgageMSR.toLocaleString('en-US', {maximumFractionDigits:0})}`;
            document.getElementById('res-max-loan').innerText = `$${Math.round(maxLoanTDSR).toLocaleString()}`;
            document.getElementById('tdsr-results').classList.remove('hidden');
        }

        function calculateProceeds() {
            const price = parseFloat(document.getElementById('proc-price').value);
            const loan = parseFloat(document.getElementById('proc-loan').value);
            const cpf = parseFloat(document.getElementById('proc-cpf').value);

            const deductions = loan + cpf;
            const proceeds = price - deductions;

            document.getElementById('res-sale-price').innerText = `$${price.toLocaleString()}`;
            document.getElementById('res-deductions').innerText = `-$${deductions.toLocaleString()}`;
            document.getElementById('res-cash-proceeds').innerText = `$${Math.max(0, proceeds).toLocaleString()}`;
            document.getElementById('proceeds-results').classList.remove('hidden');
        }

        function calculatePurchase() {
            const price = parseFloat(document.getElementById('buy-price').value);
            const profile = document.getElementById('buy-profile').value;

            // BSD Calculation
            let bsd = 0;
            if (price <= 180000) bsd = price * 0.01;
            else if (price <= 360000) bsd = 1800 + (price - 180000) * 0.02;
            else if (price <= 1000000) bsd = 5400 + (price - 360000) * 0.03;
            else bsd = 24600 + (price - 1000000) * 0.04;

            let absdRate = 0;
            if (profile === 'sc-2') absdRate = 0.20;
            if (profile === 'pr-1') absdRate = 0.05;
            const absd = price * absdRate;

            const minCashDown = price * 0.05;
            const totalUpfront = minCashDown + bsd + absd;

            document.getElementById('res-bsd').innerText = `$${Math.round(bsd).toLocaleString()}`;
            document.getElementById('res-absd').innerText = `$${Math.round(absd).toLocaleString()}`;
            document.getElementById('res-cash-down').innerText = `$${Math.round(minCashDown).toLocaleString()}`;
            document.getElementById('res-total-upfront').innerText = `$${Math.round(totalUpfront).toLocaleString()}`;
            document.getElementById('purchase-results').classList.remove('hidden');
        }

        function renderProperties(list) {

                    </div>
                    <div class="text-right font-bold text-slate-800 text-sm">${p.price}</div>
                </div>
            `).join('');
        }

        function filterProperties() {
            const query = document.getElementById('prop-search').value.toLowerCase();
            const filtered = properties.filter(p => p.name.toLowerCase().includes(query) || p.town.toLowerCase().includes(query) || p.type.toLowerCase().includes(query));
            renderProperties(filtered);
        }

        }).addTo(map);

        ];

        launches.forEach(l => {
            L.marker([l.lat, l.lng]).addTo(map)
                .bindPopup(`<b>${l.name}</b><br>Land Bid: ${l.bid}<br>Breakeven: ${l.breakeven}`);
        });
